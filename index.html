const API_BASE = 'https://api.reya.xyz/v2'; const PROXY_PREFIX = 'https://corsproxy.io/?'; let marketState = []; let currentSide = 'LONG'; async function safeFetch(url) { try { const res = await fetch(url); if (res.ok) return await res.json(); } catch (e) { // Fallback via CORS proxy if direct fetch fails } const proxyRes = await fetch(PROXY_PREFIX + encodeURIComponent(url)); if (!proxyRes.ok) throw new Error('Network error or endpoint unreachable.'); return await proxyRes.json(); } async function fetchAllData() { const loading = document.getElementById('markets-loading'); const grid = document.getElementById('markets-grid'); const banner = document.getElementById('status-banner'); loading.classList.remove('hidden'); grid.classList.add('hidden'); banner.classList.add('hidden'); try { // Fetch market definitions and market stats in parallel const [definitions, summaries] = await Promise.all([ safeFetch(`${API_BASE}/perpMarketDefinitions`).catch(() => []), safeFetch(`${API_BASE}/perpMarkets/summary`).catch(() => []) ]); // Normalize data structure const markets = Array.isArray(definitions) ? definitions : (definitions.result || definitions.markets || []); const summariesMap = {}; const summaryList = Array.isArray(summaries) ? summaries : (summaries.result || summaries.summaries || []); summaryList.forEach(s => { summariesMap[s.marketId || s.id || s.symbol] = s; }); marketState = markets.map((m, idx) => { const id = m.marketId || m.id || idx; const symbol = m.symbol || m.name || `Market ${id}`; const summary = summariesMap[id] || {}; return { id: id, symbol: symbol, markPrice: Number(summary.markPrice || m.markPrice || 100), fundingRate: Number(summary.fundingRate || m.fundingRate || 0.0001), maxLeverage: Number(m.maxLeverage || 20) }; }); // Fallback demo data if API returns empty array if (marketState.length === 0) { marketState = [ { id: 1, symbol: 'ETH-PERP', markPrice: 2650.00, fundingRate: 0.00012, maxLeverage: 20 }, { id: 2, symbol: 'BTC-PERP', markPrice: 68400.00, fundingRate: 0.00008, maxLeverage: 20 }, { id: 3, symbol: 'SOL-PERP', markPrice: 145.50, fundingRate: 0.00020, maxLeverage: 10 } ]; } renderMarkets(marketState); populateSimulatorDropdown(marketState); calculateSimulation(); loading.classList.add('hidden'); grid.classList.remove('hidden'); } catch (err) { console.error(err); loading.classList.add('hidden'); banner.className = "p-4 rounded-xl text-sm border bg-rose-950/60 border-rose-800 text-rose-300"; banner.textContent = "Failed to connect to Reya L2 node API. Displaying fallback simulation markets."; banner.classList.remove('hidden'); // Fallback render marketState = [ { id: 1, symbol: 'ETH-PERP', markPrice: 2650.00, fundingRate: 0.00012, maxLeverage: 20 }, { id: 2, symbol: 'BTC-PERP', markPrice: 68400.00, fundingRate: 0.00008, maxLeverage: 20 }, { id: 3, symbol: 'SOL-PERP', markPrice: 145.50, fundingRate: 0.00020, maxLeverage: 10 } ]; renderMarkets(marketState); populateSimulatorDropdown(marketState); calculateSimulation(); grid.classList.remove('hidden'); } } function renderMarkets(markets) { const grid = document.getElementById('markets-grid'); grid.innerHTML = markets.map(m => `
ID: ${m.id}

${m.symbol}

${m.maxLeverage}x Max Lev
Mark Price $${m.markPrice.toLocaleString(undefined, {minimumFractionDigits: 2})}
Funding Rate ${(m.fundingRate * 100).toFixed(4)}%
`).join(''); } function populateSimulatorDropdown(markets) { const select = document.getElementById('sim-market-select'); select.innerHTML = markets.map(m => ` `).join(''); } function setSide(side) { currentSide = side; const btnLong = document.getElementById('btn-side-long'); const btnShort = document.getElementById('btn-side-short'); if (side === 'LONG') { btnLong.className = "py-2 text-xs font-bold rounded-lg border bg-emerald-950/80 border-emerald-600 text-emerald-400"; btnShort.className = "py-2 text-xs font-bold rounded-lg border bg-slate-950 border-slate-800 text-slate-400"; } else { btnLong.className = "py-2 text-xs font-bold rounded-lg border bg-slate-950 border-slate-800 text-slate-400"; btnShort.className = "py-2 text-xs font-bold rounded-lg border bg-rose-950/80 border-rose-600 text-rose-400"; } calculateSimulation(); } function updateLeverageVal() { const slider = document.getElementById('sim-leverage'); document.getElementById('sim-lev-val').textContent = slider.value; } function calculateSimulation() { const marketId = document.getElementById('sim-market-select').value; const market = marketState.find(m => String(m.id) === String(marketId)) || marketState[0]; const collateral = parseFloat(document.getElementById('sim-collateral').value) || 0; const leverage = parseInt(document.getElementById('sim-leverage').value) || 1; const markPrice = market ? market.markPrice : 100; const notional = collateral * leverage; let liqPrice = 0; if (market) { const maintenanceMarginRatio = 0.05; // 5% default MMR estimation if (currentSide === 'LONG') { liqPrice = markPrice * (1 - (1 / leverage) + maintenanceMarginRatio); if (liqPrice < 0) liqPrice = 0; } else { liqPrice = markPrice * (1 + (1 / leverage) - maintenanceMarginRatio); } } document.getElementById('res-notional').textContent = `$${notional.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2})}`; document.getElementById('res-mark').textContent = `$${markPrice.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2})}`; document.getElementById('res-liq').textContent = `$${liqPrice.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2})}`; document.getElementById('res-margin').textContent = `$${collateral.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2})}`; } async function fetchWalletPositions() { const address = document.getElementById('wallet-address-input').value.trim(); const output = document.getElementById('wallet-output'); output.classList.remove('hidden'); if (!address.startsWith('0x') || address.length < 10) { output.innerHTML = `Please enter a valid Ethereum wallet address (0x...).`; return; } output.innerHTML = `Querying Reya L2 state for ${address}...`; try { const data = await safeFetch(`${API_BASE}/accounts/${address}/positions`); output.innerHTML = `
${JSON.stringify(data, null, 2)}
`; } catch (err) { output.innerHTML = `No active positions found or account uninitialized on Reya L2 for this address.`; } } // Initialize on load window.onload = fetchAllData;
